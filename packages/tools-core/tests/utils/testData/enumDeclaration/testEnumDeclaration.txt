 */

/**
 * Response state represented as a string enum declaration.
 */
export enum ResponseState {
	Success = "success",
	Failed = "failed"
}

/**
 * Numeric HTTP-like code enum declaration.
 */
export enum ResponseCode {
	Ok = 200,
	Accepted,
	Conflict = 409
}

/**
 * Test enum declaration handling.
 */
export interface TestEnumDeclaration {
	/**
	 * Enum declaration reference.
	 */
	state: ResponseState;

	/**
	 * Numeric enum declaration reference.
	 */
	code: ResponseCode;
}
